Challenges

  1. Hold the arm up until the car's been gone 3 seconds. Use millis(), not delay(3000), so the sensor keeps reading.
  2. Count 5 spots. Each time the gate opens, one car came in. The button on pin 2 is a car leaving. Print Spots left: 4 to the Serial Monitor when the number changes.
  3. At 0 spots the gate stays shut for a new car and the red light blinks. A button press frees a spot.
  4. Harder: move the arm 1 degree every 15 ms with no delay(). If a car shows up while the arm is coming down, the arm goes back up.

// Parking gate, challenge 2 solution. RCC ACM Arduino lab, 2026-10-29.
// Counts open spots. Each time the gate opens, one car came in.
// Each press of the exit button is one car going out.

#include <Servo.h>

const int TRIG = 12;       // sensor Trig
const int ECHO = 11;       // sensor Echo
const int RED = 4;         // red light
const int GREEN = 5;       // green light
const int SERVO_PIN = 9;   // servo orange wire
const int EXIT_BUTTON = 2; // exit button: one leg row to pin 2, the other to the - rail

const int ARM_DOWN = 0;    // arm angle when the gate is closed
const int ARM_UP = 90;     // arm angle when the gate is open
const int CAR_CM = 10;     // anything closer than this many cm is a car

const int SPOTS = 5;                  // spots in the lot
const unsigned long HOLD_MS = 3000;   // how long the arm stays up after the car leaves

Servo gate;

void setup() {
  Serial.begin(9600);
  pinMode(TRIG, OUTPUT);
  pinMode(ECHO, INPUT);
  pinMode(RED, OUTPUT);
  pinMode(GREEN, OUTPUT);
  pinMode(EXIT_BUTTON, INPUT_PULLUP);   // reads HIGH until pressed
  gate.attach(SERVO_PIN);
  gate.write(ARM_DOWN);
}

// Gets the distance to the nearest thing in front of the sensor, in cm.
// Gives back 0 when no echo comes back.
long readCm() {
  digitalWrite(TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG, HIGH);
  delayMicroseconds(10);                  // a 10 microsecond pulse starts a reading
  digitalWrite(TRIG, LOW);
  long us = pulseIn(ECHO, HIGH, 30000);   // wait at most 30 ms for the echo
  return us / 58;                         // 58 microseconds per cm
}

int spotsLeft = SPOTS;
unsigned long lastSeen = 0;   // millis() the last time a car was seen
bool isOpen = false;          // is the arm up right now?
int lastButton = HIGH;        // the button's reading last time through loop()

void openGate() {
  digitalWrite(GREEN, HIGH);
  digitalWrite(RED, LOW);
  gate.write(ARM_UP);
  isOpen = true;
}

void closeGate() {
  digitalWrite(RED, HIGH);
  digitalWrite(GREEN, LOW);
  gate.write(ARM_DOWN);
  isOpen = false;
}

void printSpots() {
  Serial.print("Spots left: ");
  Serial.println(spotsLeft);
}

void loop() {
  long cm = readCm();
  bool car = (cm > 0 && cm < CAR_CM);

  // Car in: count once, when the gate goes from closed to open.
  if (car) {
    lastSeen = millis();
    if (!isOpen) {
      openGate();
      spotsLeft = spotsLeft - 1;
      printSpots();
    }
  } else if (isOpen && millis() - lastSeen >= HOLD_MS) {
    closeGate();
  } else if (!isOpen) {
    closeGate();
  }

  // Car out: count once per press, when the reading goes from HIGH to LOW.
  int button = digitalRead(EXIT_BUTTON);
  if (button == LOW && lastButton == HIGH) {
    if (spotsLeft < SPOTS) {
      spotsLeft = spotsLeft + 1;
    }
    printSpots();
  }
  lastButton = button;

  delay(60);
}

// Parking gate, challenge 3 solution. RCC ACM Arduino lab, 2026-10-29.
// Full lot: at 0 spots the gate stays down and the red light blinks.

#include <Servo.h>

const int TRIG = 12;       // sensor Trig
const int ECHO = 11;       // sensor Echo
const int RED = 4;         // red light
const int GREEN = 5;       // green light
const int SERVO_PIN = 9;   // servo orange wire
const int EXIT_BUTTON = 2; // exit button: one leg row to pin 2, the other to the - rail

const int ARM_DOWN = 0;    // arm angle when the gate is closed
const int ARM_UP = 90;     // arm angle when the gate is open
const int CAR_CM = 10;     // anything closer than this many cm is a car

const int SPOTS = 5;                  // spots in the lot
const unsigned long HOLD_MS = 3000;   // how long the arm stays up after the car leaves

Servo gate;

void setup() {
  Serial.begin(9600);
  pinMode(TRIG, OUTPUT);
  pinMode(ECHO, INPUT);
  pinMode(RED, OUTPUT);
  pinMode(GREEN, OUTPUT);
  pinMode(EXIT_BUTTON, INPUT_PULLUP);   // reads HIGH until pressed
  gate.attach(SERVO_PIN);
  gate.write(ARM_DOWN);
}

// Gets the distance to the nearest thing in front of the sensor, in cm.
// Gives back 0 when no echo comes back.
long readCm() {
  digitalWrite(TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG, HIGH);
  delayMicroseconds(10);                  // a 10 microsecond pulse starts a reading
  digitalWrite(TRIG, LOW);
  long us = pulseIn(ECHO, HIGH, 30000);   // wait at most 30 ms for the echo
  return us / 58;                         // 58 microseconds per cm
}

const unsigned long BLINK_MS = 250;   // red light blink speed when the lot is full

int spotsLeft = SPOTS;
unsigned long lastSeen = 0;
bool isOpen = false;
int lastButton = HIGH;

void openGate() {
  digitalWrite(GREEN, HIGH);
  digitalWrite(RED, LOW);
  gate.write(ARM_UP);
  isOpen = true;
}

void closeGate() {
  digitalWrite(GREEN, LOW);
  gate.write(ARM_DOWN);
  isOpen = false;
}

void printSpots() {
  Serial.print("Spots left: ");
  Serial.println(spotsLeft);
}

void loop() {
  long cm = readCm();
  bool car = (cm > 0 && cm < CAR_CM);
  bool full = (spotsLeft == 0);

  if (car && (isOpen || !full)) {
    lastSeen = millis();
    if (!isOpen) {
      openGate();
      spotsLeft = spotsLeft - 1;
      printSpots();
    }
  } else if (isOpen && millis() - lastSeen >= HOLD_MS) {
    closeGate();
  }

  // Red light: off while open, blinking when full, on otherwise.
  if (!isOpen) {
    if (full) {
      bool blinkOn = (millis() / BLINK_MS) % 2 == 0;
      digitalWrite(RED, blinkOn ? HIGH : LOW);
    } else {
      digitalWrite(RED, HIGH);
    }
  }

  int button = digitalRead(EXIT_BUTTON);
  if (button == LOW && lastButton == HIGH) {
    if (spotsLeft < SPOTS) {
      spotsLeft = spotsLeft + 1;
    }
    printSpots();
  }
  lastButton = button;

  delay(60);
}
Harder: a gate that goes back up
Harder: a gate that goes back up
// Parking gate, stretch solution. RCC ACM Arduino lab, 2026-10-29.
// The arm moves 1 degree at a time with no delay(), so the sensor keeps
// reading while it moves. If a car shows up while the arm is coming down,
// the arm goes back up. Keeps the spot count and full-lot light from challenge 3.
//
// The gate is always in one of 4 states:
//   CLOSED  -> a car shows up and there's a spot -> OPENING
//   OPENING -> arm reaches ARM_UP                 -> OPEN
//   OPEN    -> no car for HOLD_MS                 -> CLOSING
//   CLOSING -> arm reaches ARM_DOWN               -> CLOSED
//   CLOSING -> a car shows up                     -> OPENING (no new count)

#include <Servo.h>

const int TRIG = 12;
const int ECHO = 11;
const int RED = 4;
const int GREEN = 5;
const int SERVO_PIN = 9;
const int EXIT_BUTTON = 2;

const int ARM_DOWN = 0;
const int ARM_UP = 90;
const int CAR_CM = 10;

const int SPOTS = 5;
const unsigned long HOLD_MS = 3000;
const unsigned long STEP_MS = 15;     // one degree every 15 ms, like Sweep
const unsigned long SENSOR_MS = 60;   // one reading every 60 ms (datasheet)
const unsigned long BLINK_MS = 250;

enum GateState { CLOSED, OPENING, OPEN, CLOSING };
GateState state = CLOSED;

Servo gate;
int angle = ARM_DOWN;
int spotsLeft = SPOTS;
bool car = false;
unsigned long lastSeen = 0;
unsigned long lastStep = 0;
unsigned long lastRead = 0;
int lastButton = HIGH;

void setup() {
  Serial.begin(9600);
  pinMode(TRIG, OUTPUT);
  pinMode(ECHO, INPUT);
  pinMode(RED, OUTPUT);
  pinMode(GREEN, OUTPUT);
  pinMode(EXIT_BUTTON, INPUT_PULLUP);
  gate.attach(SERVO_PIN);
  gate.write(angle);
}

long readCm() {
  digitalWrite(TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG, LOW);
  long us = pulseIn(ECHO, HIGH, 30000);
  return us / 58;
}

void printState() {
  const char* names[] = { "CLOSED", "OPENING", "OPEN", "CLOSING" };
  Serial.print(names[state]);
  Serial.print("   spots left: ");
  Serial.println(spotsLeft);
}

void setState(GateState next) {
  state = next;
  printState();
}

void loop() {
  unsigned long now = millis();

  // 1. Read the sensor every SENSOR_MS.
  if (now - lastRead >= SENSOR_MS) {
    lastRead = now;
    long cm = readCm();
    car = (cm > 0 && cm < CAR_CM);
    if (car) {
      lastSeen = millis();
    }
  }

  // 2. Exit button: one car out per press.
  int button = digitalRead(EXIT_BUTTON);
  if (button == LOW && lastButton == HIGH) {
    if (spotsLeft < SPOTS) {
      spotsLeft = spotsLeft + 1;
    }
    printState();
  }
  lastButton = button;

  // 3. Decide the next state.
  switch (state) {
    case CLOSED:
      if (car && spotsLeft > 0) {
        spotsLeft = spotsLeft - 1;
        setState(OPENING);
      }
      break;
    case OPENING:
      if (angle >= ARM_UP) {
        setState(OPEN);
      }
      break;
    case OPEN:
      if (!car && millis() - lastSeen >= HOLD_MS) {
        setState(CLOSING);
      }
      break;
    case CLOSING:
      if (car) {
        setState(OPENING);          // same car or a new one: don't drop the arm on it
      } else if (angle <= ARM_DOWN) {
        setState(CLOSED);
      }
      break;
  }

  // 4. Move the arm 1 degree per STEP_MS. The while catches up if
  //    pulseIn() held up the loop for a few ms.
  while (millis() - lastStep >= STEP_MS) {
    lastStep += STEP_MS;
    if (state == OPENING && angle < ARM_UP) {
      angle = angle + 1;
    } else if (state == CLOSING && angle > ARM_DOWN) {
      angle = angle - 1;
    }
  }
  gate.write(angle);

  // 5. Lights: green only when fully open. Red blinks when full and closed.
  digitalWrite(GREEN, state == OPEN ? HIGH : LOW);
  if (state == OPEN) {
    digitalWrite(RED, LOW);
  } else if (state == CLOSED && spotsLeft == 0) {
    digitalWrite(RED, (millis() / BLINK_MS) % 2 == 0 ? HIGH : LOW);
  } else {
    digitalWrite(RED, HIGH);
  }
}
